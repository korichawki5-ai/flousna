### مجموعة الهوية والأدوار
```sql
-- 1. profiles: امتداد لـ auth.users
CREATE TABLE public.profiles (
  id                 UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  display_name       TEXT,                        -- اختياري (لا اسم حقيقي مطلوب)
  wilaya_code        SMALLINT,                    -- اختياري (1..58)
  occupation         TEXT,                        -- اختياري
  preferred_locale   TEXT NOT NULL DEFAULT 'ar_DZ' CHECK (preferred_locale IN ('ar_DZ','fr_DZ')),
  theme_mode         TEXT NOT NULL DEFAULT 'system' CHECK (theme_mode IN ('light','dark','system')),
  fiscal_anchor_day  SMALLINT NOT NULL DEFAULT 1 CHECK (fiscal_anchor_day BETWEEN 1 AND 28),
  budget_mode        TEXT NOT NULL DEFAULT 'monthly' CHECK (budget_mode IN ('monthly','weekly','both')),
  base_currency      CHAR(3) NOT NULL DEFAULT 'DZD',
  trial_started_at   TIMESTAMPTZ,
  trial_ends_at      TIMESTAMPTZ,
  onboarded_at       TIMESTAMPTZ,
  created_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at         TIMESTAMPTZ                  -- soft delete (حق المحو)
);

-- 2. user_roles: الأدوار (تُتحقق داخل سياسات RLS نفسها، لا في الواجهة)
CREATE TABLE public.user_roles (
  user_id    UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  role       TEXT NOT NULL DEFAULT 'user' CHECK (role IN ('user','admin')),
  granted_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  granted_by UUID REFERENCES auth.users(id)
);
```

### مجموعة الاشتراكات
```sql
-- 3. plans: الباقات (تُدار مركزياً — تغيير الأسعار بلا تحديث التطبيق)
CREATE TABLE public.plans (
  id            TEXT PRIMARY KEY,                 -- 'individual'|'couple'|'family'
  name_ar       TEXT NOT NULL,
  name_fr       TEXT NOT NULL,
  max_profiles  SMALLINT NOT NULL,                -- 1 | 2 | 6
  max_devices   SMALLINT NOT NULL,                -- 1 | 2 | 3
  price_monthly NUMERIC(10,2) NOT NULL CHECK (price_monthly >= 0),
  price_yearly  NUMERIC(10,2) NOT NULL CHECK (price_yearly >= 0),
  is_anchor     BOOLEAN NOT NULL DEFAULT false,   -- المرساة السعرية (العائلة)
  sort_order    SMALLINT NOT NULL,
  is_active     BOOLEAN NOT NULL DEFAULT true,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 4. subscriptions
CREATE TABLE public.subscriptions (
  id                   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id              UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  plan_id              TEXT NOT NULL REFERENCES public.plans(id),
  status               TEXT NOT NULL DEFAULT 'trialing'
                       CHECK (status IN ('trialing','active','grace','past_due','canceled')),
  billing_cycle        TEXT NOT NULL CHECK (billing_cycle IN ('monthly','yearly')),
  current_period_start TIMESTAMPTZ,
  current_period_end   TIMESTAMPTZ,
  grace_ends_at        TIMESTAMPTZ,               -- ⭐ 72 ساعة سماح أوفلاين
  cancellation_reason  TEXT,
  activated_by         UUID REFERENCES public.profiles(id),
  created_at           TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at           TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (user_id)                                  -- اشتراك واحد نشط لكل مستخدم
);

-- 5. activation_requests: سجل طلبات التفعيل (تدقيق)
CREATE TABLE public.activation_requests (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id       UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  request_code  TEXT NOT NULL UNIQUE,              -- FL-XXXX-XXXX (8 محارف آمنة)
  plan_id       TEXT NOT NULL REFERENCES public.plans(id),
  billing_cycle TEXT NOT NULL CHECK (billing_cycle IN ('monthly','yearly')),
  amount_dzd    NUMERIC(10,2) NOT NULL,
  status        TEXT NOT NULL DEFAULT 'pending'
                CHECK (status IN ('pending','verified','activated','rejected','expired')),
  admin_note    TEXT,                              -- ⚠️ ممنوع بيانات مصرفية هنا
  requested_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  processed_at  TIMESTAMPTZ,
  processed_by  UUID REFERENCES public.profiles(id)
);
-- ⭐ Rate limiting على مستوى القاعدة: 3 طلبات/ساعة/مستخدم
CREATE INDEX idx_actreq_user_time ON public.activation_requests(user_id, requested_at DESC);

-- 6. devices: ربط الأجهزة (لباقات زوجان/عائلة)
CREATE TABLE public.devices (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id       UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  device_id     UUID NOT NULL,                     -- device_install_id محلي
  platform      TEXT NOT NULL CHECK (platform IN ('android','windows','web','ios')),
  app_version   TEXT,
  last_seen_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  is_active     BOOLEAN NOT NULL DEFAULT true,
  UNIQUE (user_id, device_id)
);
```

## 2.3 سياسات RLS (المبدأ — تُكتب كاملة في المرحلة 3)

```sql
-- ⭐ RLS مفعّل على كل الجداول بدون استثناء
ALTER TABLE public.profiles            ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.transactions        ENABLE ROW LEVEL SECURITY;
-- … (كل الـ23 جدولاً)

-- دالة مساعدة: هل المستخدم أدمن؟ (تُستخدم داخل السياسات نفسها)
CREATE OR REPLACE FUNCTION public.is_admin() RETURNS BOOLEAN
LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.user_roles
    WHERE user_id = auth.uid() AND role = 'admin'
  );
$$;

-- مثال 1: المستخدم يرى بياناته فقط
CREATE POLICY "tx_select_own" ON public.transactions
  FOR SELECT USING (auth.uid() = user_id OR public.is_admin());

CREATE POLICY "tx_insert_own" ON public.transactions
  FOR INSERT WITH CHECK (auth.uid() = user_id);

CREATE POLICY "tx_update_own" ON public.transactions
  FOR UPDATE USING (auth.uid() = user_id OR public.is_admin())
             WITH CHECK (auth.uid() = user_id);

-- مثال 2: ⚠️ لا سياسة INSERT/UPDATE على subscriptions للمستخدم أبداً
--   → التفعيل حصري عبر Edge Function بمفتاح Service Role
--   → هذا يمنع المستخدم من تفعيل نفسه بتعديل القاعدة مباشرة
CREATE POLICY "sub_select_own" ON public.subscriptions
  FOR SELECT USING (auth.uid() = user_id OR public.is_admin());
-- (لا INSERT ولا UPDATE ولا DELETE للمستخدم — عن قصد)

-- مثال 3: consent_logs — قراءة فقط للمستخدم، لا تعديل إطلاقاً
CREATE POLICY "consent_select_own" ON public.consent_logs
  FOR SELECT USING (auth.uid() = user_id OR public.is_admin());
CREATE POLICY "consent_insert_own" ON public.consent_logs
  FOR INSERT WITH CHECK (auth.uid() = user_id OR anonymous_id IS NOT NULL);
-- (لا UPDATE · لا DELETE — Append-only)

-- مثال 4: الأدوار — المستخدم لا يستطيع ترقية نفسه
CREATE POLICY "roles_select_own" ON public.user_roles
  FOR SELECT USING (auth.uid() = user_id);
-- (لا INSERT/UPDATE/DELETE للمستخدم — الإدارة عبر Edge Function فقط)

-- مثال 5: app_versions — عام للقراءة (للتحديثات، حتى بدون حساب)
CREATE POLICY "versions_public_read" ON public.app_versions
  FOR SELECT USING (true);
```

### ⚠️ ثغرة معروفة سأصرّح بها بصدق
> **`is_admin()` تستعمل `SECURITY DEFINER`** — وهي ضرورية لعمل السياسات، لكنها **نقطة حساسة**.
> **التخفيف:** `SET search_path = public` (يمنع hijacking) + `STABLE` + لا تُمنح صلاحية `admin` إلا عبر Edge Function بمفتاح Service Role + سجل تدقيق لكل منح.
> **لا أدّعي حماية 100%.** سأراجع السياسات يدوياً سطراً بسطر في المرحلة 4 وأسلّمك تقرير المراجعة.

## 3.2 تدفقات المستخدم الأساسية (User Flows)

### 🅰️ الرحلة 1 — أول تشغيل (الأهم: لا تفقد المستخدم هنا)
```
Splash (≤2s)
  → Onboarding 1: «اعرف فين راحت فلوسك» [تخطي متاح]
  → Onboarding 2: «ميزانية تناسب دخلك — يومي أو شهري»
  → Onboarding 3: «بياناتك في هاتفك. لا تُرسل إلا بطلبك» ⭐ (حجة قانونية + تسويقية)
  → ⚖️ الموافقة #1 (حاجز — صندوق غير مُعلَّم — رفض = خروج)
  → اختيار المسار: [جرّب بدون حساب] / [إنشاء حساب]
  → الإعداد الأولي (3 خطوات · كلها قابلة للتخطي)
  → 🏠 الرئيسية بحالة «فارغة» موجَّهة:
       «لا توجد حركات بعد. أضف أول مصروف لتبدأ 👇» + سهم نحو FAB
```
**الهدف:** من التنزيل إلى أول حركة مسجّلة في **≤ 90 ثانية**

### 🅱️ الرحلة 2 — تسجيل مصروف (الوظيفة الأكثر تكراراً)
```
FAB ➕ (من أي شاشة)
  → [مصروف] محدَّد افتراضياً
  → اكتب المبلغ (لوحة أرقام كبيرة · فواصل تظهر مباشرة)
  → اختر الفئة (شبكة — الأكثر استخداماً في الأعلى)
  → 💾 حفظ
  → ✅ Snackbar: «تم تسجيل 1.250 دج في الطعام» + عودة فورية
```
**الهدف: 3 نقرات كحد أقصى** · **يعمل 100% بلا إنترنت**

### 🅲 الرحلة 3 — تجاوز حد الميزانية
```
إشعار محلي: «⚠️ اقتربت من حدّ الطعام (80%)»
  → فتح التطبيق → 🏠 الرئيسية تُظهر شريطاً كهرمانياً
  → 📊 الميزانية → فئة الطعام بلون كهرماني + «بقي 1.400 دج»
  → اقتراح: «أكبر مصروفين هذا الأسبوع: … و …»
  → [تعديل الحد] أو [متابعة]
```

### 🅳 الرحلة 4 — الاشتراك عبر واتساب ⭐ (رحلة الإيراد)
```
اليوم 5 من التجربة
  → 🏠 شريط علوي: «⏳ بقي يومان من التجربة المجانية»
اليوم 8
  → إضافة حركة → 🔒 شاشة انتهاء التجربة (قفل جزئي + عد تنازلي)
     «انتهت فترة التجربة. سجّلتَ 47 حركة — بياناتك محفوظة وآمنة.»
  → الباقات (العائلة أولاً = المرساة · السنوي افتراضي · «وفّر 30%»)
  → [🟢 أكمل عبر واتساب]  ← CTA واحد فقط
  → يولَّد رمز الطلب FL-7K3M-9QXR (Edge Function + rate limit)
  → يفتح واتساب برسالة معبّأة (بلا أي بيانات مصرفية)
  → المستخدم يحوّل ويرسل الوصل + رقم العملية
  → التطبيق يعرض: «📨 طلبك قيد المعالجة — نفعّله خلال ≤ 24 ساعة»
  → 🛡️ الأدمن: يتحقق → [تفعيل]
  → Edge Function (Service Role) تحدّث subscriptions
  → Realtime → «🎉 تم تفعيل Pro — شكراً لك!» + إيصال PDF
```

### 🅴 الرحلة 5 — الأدمن يفعّل اشتراكاً
```
🛡️ لوحة الإدارة → طلبات التفعيل → الطابور (الأقدم أولاً)
  → تفاصيل الطلب: الرمز · المستخدم · الباقة · المبلغ · التاريخ
  → تتحقق في تطبيقك المصرفي من رقم العملية
  → [✅ تفعيل] → اختر المدة (شهر/سنة) → تأكيد
  → Edge Function: UPDATE subscriptions + INSERT audit_logs
  → 📧 بريد تأكيد للمستخدم (اختياري)
  → أو [❌ رفض] → سبب (يظهر للمستخدم بلطف)
```

### 🅵 الرحلة 6 — الحق في المحو (قانوني — إلزامي)
```
⚖️ القانوني → 🗑️ حذف حسابي وبياناتي نهائياً
  → تحذير صريح: «سيُحذف كل شيء ولا يمكن التراجع»
  → قائمة ما سيُحذف (شفافية): الحركات · الميزانيات · الديون · الأهداف · الملف
  → تأكيد بالكتابة: «احذف» (أو عدّ 10 ثوانٍ)
  → Edge Function: حذف CASCADE من PostgreSQL + Storage + إشعار بحذف المحلي
  → 📧 تأكيد بالبريد + سجل في audit_logs
  → التطبيق يعود إلى حالة «أول تشغيل»
```

### 🅶 الرحلة 7 — الأوفلاين ثم المزامنة
```
📵 بلا إنترنت
  → كل شيء يعمل (إضافة · عرض · تقارير)
  → مؤشر صغير أعلى الشاشة: «📵 غير متصل — 3 حركات بانتظار المزامنة»
📶 عودة الاتصال
  → المزامنة تلقائياً في الخلفية
  → «✅ تمت مزامنة 3 حركات»
⚠️ تعارض (نفس الحركة عُدّلت من جهازين)
  → Last-Write-Wins + النسخة الأخرى محفوظة في سجل التعديلات (لا تُفقد)
  → إشعار: «تم دمج تغييرات متعارضة — راجع سجل التعديلات»
```

# ⚠️ الجزء 7 — الحدود المعروفة (بصدق كامل، بلا ادّعاء)

```
1.  🔴 لا أدّعي حماية 100%. RLS قوية لكن أي نظام قابل للاختراق.
2.  🔴 is_admin() تستعمل SECURITY DEFINER — نقطة حساسة (مخفَّفة بـ search_path ثابت).
3.  🔴 Supabase المجاني يتوقف بعد 7 أيام — الحل: ping يومي (مؤقت) → Pro عند الجدّية.
4.  🔴 Supabase المجاني بلا نسخ احتياطي آلي — الحل: نسخة أسبوعية مجدولة (سأبنيها).
5.  🔴 البيانات في ألمانيا = نقل عبر الحدود → يتطلب ترخيص ANPDP (لا أستطيع فعله عنك).
6.  🔴 منع إعادة ضبط التجربة: كامل مع حساب · جزئي بدون حساب (لا حل تقني مثالي).
7.  🟡 Last-Write-Wins في المزامنة: لا يُفقد شيء، لكن النسخة الظاهرة هي الأحدث.
8.  🟡 الإشعارات: محلية فقط في MVP (تعمل بلا FCM) · Push في المرحلة 5 وقد لا يعمل
    على أجهزة بلا Google Play Services.
9.  🟡 .exe ويندوز غير موقّع → تحذير SmartScreen (دليل مصوّر في الموقع).
10. 🟡 الحساب الفلكي للهجري ≠ الإعلان الرسمي (فرق يوم أو يومين) → تصحيح يدوي.
11. 🟡 حاسبة الزكاة والمبالغ الموسمية إرشادية — ليست فتوى ولا مصدر أسعار.
12. 🟡 التفعيل اليدوي عبر واتساب عملي حتى ~50 عملية/يوم — بعدها يلزم Satim/PSP.
13. 🔴 لستُ محامياً. كل ما يتعلق بالامتثال يحتاج مراجعة قانونية حقيقية.
14. 🟡 ثنائية اللغة + العملات + النظام الموسمي = مشروع كبير. ضبطتُ نطاق MVP
    بصرامة (22 وظيفة) — وأي إضافة قبل الإطلاق تؤخّره.
```
